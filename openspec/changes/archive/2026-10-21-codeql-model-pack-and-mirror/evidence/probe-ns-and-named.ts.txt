import Fastify from "fastify";
import fs from "node:fs";
import { isAllowedCwd } from "@blackbelt-technology/pi-dashboard-shared/cwd-guard.js";
import * as G from "@blackbelt-technology/pi-dashboard-shared/cwd-guard.js";
const app = Fastify();
const known = () => ["/a"];
app.get("/named", async (req: any) => { const c = req.query.cwd; if (isAllowedCwd(c, known)) { return fs.readFileSync(c + "/x"); } return null; });
app.get("/ns", async (req: any) => { const c = req.query.cwd; if (G.isAllowedCwd(c, known)) { return fs.readFileSync(c + "/y"); } return null; });
app.listen({ port: 1 });
